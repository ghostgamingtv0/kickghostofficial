document.getElementById("connectWallet").onclick = async () => { let provider = null; // التأكد من اختيار MetaMask فقط وتجنب الصراعات if (typeof window.ethereum !== 'undefined') { if (window.ethereum.providers && window.ethereum.providers.length) { // إذا كان هناك أكثر من محفظة (مثل Phantom و MetaMask) provider = window.ethereum.providers.find((p) => p.isMetaMask); } else if (window.ethereum.isMetaMask) { provider = window.ethereum; } else { provider = window.ethereum; // محفظة أخرى متوافقة } } if (provider) { try { const web3Provider = new ethers.providers.Web3Provider(provider); // طلب الحسابات const accounts = await web3Provider.send("eth_requestAccounts", []); const signer = web3Provider.getSigner(); const addr = accounts[0]; // توثيق المحفظة برسالة await signer.signMessage(`KICKGHOST Verification\nWallet: ${addr}`); document.getElementById("connectWallet").innerText = "Verified / تم التوثيق ✅"; document.getElementById("connectWallet").disabled = true; document.getElementById("claimStep").style.display = "block"; } catch (err) { console.error("User denied or error:", err); alert("خطأ: تأكد من قبول العملية في MetaMask واختيار شبكة Polygon"); } } else { alert("لم يتم العثور على MetaMask. يرجى تثبيتها أو إغلاق محافظ Solana مؤقتاً."); } };
